the API and the UI become ready:

  kubectl -n {{ .Release.Namespace }} rollout status statefulset/{{ include "apgar.fullname" . }}-worker
  kubectl -n {{ .Release.Namespace }} rollout status deployment/{{ include "apgar.fullname" . }}-api

{{ include "apgar.secretName" . }}.

  kubectl -n {{ .Release.Namespace }} port-forward service/{{ include "apgar.fullname" . }}-ui 8080:{{ .Values.ui.service.port }}

{{ include "apgar.secretName" . }}.
